// Единая настройка видимости и отметок незаполненного вместо трёх копий
// по ~100 строк в исходном модуле.
//
// Параметры:
//   ТипЛица               - Строка - "ЮрЛицо" / "ФизЛицо" / "" (не выбрано);
//   ПоказыватьРабочиеБлоки - Булево - показывать ли блоки банка/подвала для ЮЛ
//                            (для ЮЛ они открываются после «Заполнить реквизиты»).
&НаКлиенте
Процедура НастроитьЭлементыПоТипуЛица(ТипЛица, ПоказыватьРабочиеБлоки)

	ЭтоЮЛ = (ТипЛица = "ЮрЛицо");
	ЭтоФЛ = (ТипЛица = "ФизЛицо");

	Элементы.ДанныеЮЛ.Видимость = ЭтоЮЛ И ПоказыватьРабочиеБлоки;
	Элементы.ДанныеФЛ.Видимость = ЭтоФЛ;

	ПоказыватьБанк = ЭтоФЛ Или (ЭтоЮЛ И ПоказыватьРабочиеБлоки);
	Элементы.ШапкаБанкСчет.Видимость = ПоказыватьБанк;
	Элементы.Подвал.Видимость = ПоказыватьБанк;
	Элементы.ГруппаСоздатьКарточкуПартнёра.Видимость = ПоказыватьБанк;

	Элементы.Email.Видимость = ЭтоЮЛ;
	Элементы.Телефон.Видимость = ЭтоЮЛ;
	Элементы.ТелефонКИФизЛица.Видимость = ЭтоФЛ;
	Элементы.НаименованиеКИЭлектроннаяПочта.Видимость = ЭтоФЛ;
	Элементы.Город.Видимость = ЭтоФЛ;

	Элементы.ЗаполнитьРеквизиты.Доступность = ЭтоЮЛ;

	УстановитьОтметкиНезаполненного("Email, АгентскийНДС, СтавкаНДС", ЭтоЮЛ);
	УстановитьОтметкиНезаполненного(
		"Фамилия, Имя, Отчество,
		|ДокументВид, ДокументСерия, ДокументНомер, ДокументДатаВыдачи, ДокументКемВыдан, ДокументКодПодразделения,
		|Индекс, Адрес, Страна, ДатаРождения, Город, НаименованиеКИЭлектроннаяПочта", ЭтоФЛ);

&НаКлиенте
Процедура ИННИзменениеТекстаРедактирования(Элемент, Текст, СтандартнаяОбработка)

	тИНН = СокрЛП(Текст);
	тДлинаСтроки = СтрДлина(тИНН);
	ТипЛица = ТекущийТипЛица();

	ПоказатьЭлементыИП(Ложь);

	Если ТипЛица = "ФизЛицо" И тДлинаСтроки = 12 Тогда
		ЗаполнитьОбъектДляФизЛица(тИНН);
	КонецЕсли;

	Если (тДлинаСтроки = 10 ИЛИ тДлинаСтроки = 12) И ТипЛица <> "" Тогда

		Объект.ИНН = тИНН;

		тКонтрагент = НайтиКонтрагентаПоИНН(тИНН);
		Если тКонтрагент <> Неопределено Тогда
			Сообщить("По введённому ИНН в базе найден контрагент");
			Возврат;
		КонецЕсли;

		Если ТипЛица = "ЮрЛицо" Тогда

			Если Не ЕстьДанныеВЕГРЮЛ(тИНН) Тогда
				ПоказатьПредупреждение(,
					"Проверьте правильность введённого номера",
					, "ИНН не найден в базе ЕГРЮЛ");
				Возврат;
			КонецЕсли;

			Если тДлинаСтроки = 12 Тогда // ИП
				ПоказатьЭлементыИП(Истина);
			КонецЕсли;

		КонецЕсли;

	КонецЕсли;

// Заполняет документ и связанные справочники данными по ИНН (ЮЛ или ИП).
// Все обращения к внешним сервисам (гос. реестры, ЕГРЮЛ, склонение ФИО)
// выполняются ДО открытия транзакции.
&НаСервере
Функция ЗаполнитьРеквизитыНаСервере(тИНН)

	Объект.ВидАдреса = Справочники.ВидыАдресов.ЮридическийАдрес;
	Объект.ИНН = Сервисный.УдалитьИзСтрокиНеЦифры(тИНН);

	Если Не ЗначениеЗаполнено(Объект.ИНН) Тогда
		Сообщить("После очистки ИНН оказался пустым");
		Возврат Ложь;
	КонецЕсли;

	ДлинаИНН = СтрДлина(Объект.ИНН);
	Если ДлинаИНН <> 10 И ДлинаИНН <> 12 Тогда
		Сообщить("Проверьте длину ИНН");
		Возврат Ложь;
	КонецЕсли;

	ЭтоИП = (ДлинаИНН = 12);

	нКонтрагент = НайтиКонтрагентаПоИНН(Объект.ИНН);

	// --- Внешние сервисы: до транзакции ---

	РеквизитыГосРеестра = Неопределено;
	Если нКонтрагент = Неопределено Тогда
		РеквизитыГосРеестра = ПолучитьРеквизитыИзГосРеестра();
		Если РеквизитыГосРеестра = Неопределено Или Не РеквизитыГосРеестра.Выполнено Тогда
			Сообщить("Не удалось получить данные по ИНН " + ?(ЭтоИП, "ИП", "юридического лица"));
			Возврат Ложь;
		КонецЕсли;
	КонецЕсли;

	СтруктураОтвета = ПолучитьОтветЕГРЮЛ(Объект.ИНН);

	Если ЭтоИП Тогда
		ДанныеЕГРЮЛ = РазобратьОтветЕГРЮЛДляИП(СтруктураОтвета);
		ФИОРодПадеж = ФИОвРодительномПадеже(Объект.Фамилия, Объект.Имя, Объект.Отчество);
	Иначе
		ДанныеЕГРЮЛ = РазобратьОтветЕГРЮЛДляЮЛ(СтруктураОтвета);
		ФИОРодПадеж = ФИОвРодительномПадеже(ДанныеЕГРЮЛ.Фамилия, ДанныеЕГРЮЛ.Имя, ДанныеЕГРЮЛ.Отчество);
	КонецЕсли;

	// --- Запись данных: одна транзакция ---

	НачатьТранзакцию();
	Попытка

		Если нКонтрагент = Неопределено Тогда
			Объект.КонтрагентНаименованиеДляДокументов = РеквизитыГосРеестра.Реквизиты.НаименованиеДляДокументов;
			нКонтрагент = СоздатьКонтрагента(РеквизитыГосРеестра.Реквизиты);
		КонецЕсли;

		Если ЭтоИП Тогда
			ЗаполнитьАдресИП(ДанныеЕГРЮЛ, нКонтрагент);
			тКонтакт = НайтиИлиСоздатьКонтактноеЛицо(Объект.Фамилия, Объект.Имя, Объект.Отчество,
				ДанныеЕГРЮЛ.Пол, "", ФИОРодПадеж);
		Иначе
			СоздатьЮридическийАдрес(ДанныеЕГРЮЛ, нКонтрагент);
			тКонтакт = НайтиИлиСоздатьКонтактноеЛицо(ДанныеЕГРЮЛ.Фамилия, ДанныеЕГРЮЛ.Имя, ДанныеЕГРЮЛ.Отчество,
				ДанныеЕГРЮЛ.Пол, ДанныеЕГРЮЛ.Должность, ФИОРодПадеж);
		КонецЕсли;

		Объект.ОсновноеКонтактноеЛицо = тКонтакт;
		Объект.ОтветственноеЛицоЗаДоставку = тКонтакт;

		СоздатьКИКонтрагента(нКонтрагент);
		ОбновитьНДСКонтрагента(нКонтрагент);
		ЗаполнитьОбъектДаннымиКонтрагента(Объект.ИНН);

		ЗафиксироватьТранзакцию();

	Исключение
		ОтменитьТранзакцию();
		ТекстОшибки = ОписаниеОшибки();
		Сообщить("- Ошибка при заполнении реквизитов: " + ТекстОшибки);
		ЗаписатьОшибкуВЖурнал("ЗаполнитьРеквизитыНаСервере", ТекстОшибки);
		Возврат Ложь;
	КонецПопытки;

	ЗаписатьИнформациюВЖурнал("ЗаполнитьРеквизитыНаСервере",
		"Данные заполнены по ИНН " + Объект.ИНН);

	Возврат Истина;

	Если Не ПроверитьЗаполнение() Тогда
		Возврат Ложь;
	КонецЕсли;

	длинаИНН = СтрДлина(Объект.ИНН);
	ЭтоЮрЛицо = (Объект.ЮрФизЛицо = Перечисления.ЮридическоеФизическоеЛицо.ЮридическоеЛицо);
	ЭтоСамозанятый = (длинаИНН = 12 И Объект.ЮрФизЛицо = Перечисления.ЮридическоеФизическоеЛицо.ФизическоеЛицо);
	ЭтоИП = (длинаИНН = 12 И ЭтоЮрЛицо);

	Если ЭтоСамозанятый Тогда

		// Склонение ФИО (внешний вызов) — до транзакции
		Если Не ЗначениеЗаполнено(Объект.ФИОРодительныйПадеж) Тогда
			Объект.ФИОРодительныйПадеж = ФИОвРодительномПадеже(Объект.Фамилия, Объект.Имя, Объект.Отчество);
		КонецЕсли;

		НачатьТранзакцию();
		Попытка

			Если Не ЕстьФизЛицоВБазе(Объект.ИНН) Тогда
				СоздатьФизЛицо(Истина);
			КонецЕсли;

			Объект.ФизЛицо = ЗаполнитьОбъектДаннымиФизЛица();
			ПеренестиСвязанныеОбъектыНаКонтрагента(Объект.ФизЛицо);

			ЗафиксироватьТранзакцию();

		Исключение
			ОтменитьТранзакцию();
			ТекстОшибки = ОписаниеОшибки();
			Сообщить("- Ошибка при создании реквизитов: " + ТекстОшибки);
			ЗаписатьОшибкуВЖурнал("СоздатьКарточкуПартнераНаСервере", ТекстОшибки);
			Возврат Ложь;
		КонецПопытки;

		ЗаписатьИнформациюВЖурнал("СоздатьКарточкуПартнераНаСервере",
			"Создана карточка самозанятого по ИНН " + Объект.ИНН);

	КонецЕсли;

	нКонтрагент = НайтиКонтрагентаПоИНН(Объект.ИНН);

	// Дозаписываем город в объект из адреса доставки
	Если Не ЗначениеЗаполнено(Объект.Город) И ЗначениеЗаполнено(Объект.ОсновнойАдресДоставки)
		И ЗначениеЗаполнено(Объект.ОсновнойАдресДоставки.Город) Тогда

	КонецЕсли;

	// Дозаписываем город в адрес (только для ИП)
	Если ЭтоИП И ЗначениеЗаполнено(Объект.Город) И ЗначениеЗаполнено(Объект.ОсновнойАдресДоставки) Тогда
		тОбъектАдрес = Объект.ОсновнойАдресДоставки.ПолучитьОбъект();
		Если тОбъектАдрес.Город <> Объект.Город Тогда
			тОбъектАдрес.Город = Объект.Город;
			тОбъектАдрес.Записать();
		КонецЕсли;
	КонецЕсли;

	// Банковский счёт (ЮЛ и ИП)
	Если ЭтоЮрЛицо И нКонтрагент <> Неопределено
		И ЗначениеЗаполнено(Объект.БИК) И ЗначениеЗаполнено(Объект.НомерСчета) Тогда

		Счет = НайтиБанковскийСчетПоРеквизитам(нКонтрагент, Объект.БИК, Объект.НомерСчета);
		Если Счет = Неопределено Тогда
			Счет = СоздатьБанковскийСчет(нКонтрагент);
		КонецЕсли;

		ЗаполнитьРеквизитыИзБанковскогоСчета(Счет);
		Объект.ОсновнойБанковскийСчет = Счет;

		нКонтрагентОбъект = нКонтрагент.ПолучитьОбъект();
		нКонтрагентОбъект.ОсновнойБанковскийСчет = Счет;
		нКонтрагентОбъект.Записать();

	КонецЕсли;

	// Контактное лицо
	тКонтакт = НайтиИлиСоздатьКонтактноеЛицо(Объект.Фамилия, Объект.Имя, Объект.Отчество,
		Объект.Пол, "", Объект.ФИОРодительныйПадеж);
	Объект.ОсновноеКонтактноеЛицо = тКонтакт;
	Объект.ОтветственноеЛицоЗаДоставку = тКонтакт;

	Если длинаИНН = 12 И тКонтакт <> Неопределено Тогда
		ЗаполнитьФИОИзКонтакта(тКонтакт);
	КонецЕсли;

	// Контактная информация и НДС контрагента
	Если ЭтоЮрЛицо И нКонтрагент <> Неопределено Тогда
		СоздатьКИКонтрагента(нКонтрагент);
		ОбновитьНДСКонтрагента(нКонтрагент);
	КонецЕсли;

	// Дозаполняем данные физлица для ИП
	Если ЭтоИП Тогда
		ДозаполнитьФизЛицоДляИП();
	КонецЕсли;

	// Итоговая дозапись контрагента
	Если нКонтрагент <> Неопределено Тогда
		нКонтрагентОбъект = нКонтрагент.ПолучитьОбъект();
		нКонтрагентОбъект.ОсновноеКонтактноеЛицо = Объект.ОсновноеКонтактноеЛицо;
		нКонтрагентОбъект.ОтветственноеЛицоЗаДоставку = Объект.ОсновноеКонтактноеЛицо;
		нКонтрагентОбъект.ОсновнойАдресДоставки = Объект.ОсновнойАдресДоставки;
		нКонтрагентОбъект.ОсновнойБанковскийСчет = Объект.ОсновнойБанковскийСчет;
		нКонтрагентОбъект.Записать();
	КонецЕсли;

	Возврат Истина;

Процедура СоздатьДоговор(Команда)

	Если Не ЗначениеЗаполнено(Объект.УсловияДоговора) Тогда
		Сообщить("Условие договора не заполнено!");
		Возврат;
	КонецЕсли;

	Если Не ЗначениеЗаполнено(Объект.ПроцентИлиСуммаУсловия) Тогда
		Сообщить("Размер не заполнен!");
		Возврат;
	КонецЕсли;

	Если Не Записать() Тогда
		Возврат;
	КонецЕсли;

	тКонтрагентСсылка = НайтиКонтрагентаПоИНН(СокрЛП(Объект.ИНН));
	Если тКонтрагентСсылка = Неопределено Тогда
		ПоказатьПредупреждение(, "Данный контрагент удален или помечен на удаление",, "Контрагент не найден");
		Возврат;
	КонецЕсли;

	ДействующийДоговор = НайтиДействующийОткрытыйДоговор(СокрЛП(Объект.ИНН));
	Если ДействующийДоговор <> Неопределено Тогда
		Сообщить("У контрагента есть открытый договор! Обратитесь к сотруднику ОУ"
			+ Символы.ПС + "- " + Строка(ДействующийДоговор));
		Возврат;
	КонецЕсли;

	тДоговор = СоздатьДоговорНаСервере(тКонтрагентСсылка);
	Если тДоговор = Неопределено Тогда
		ПоказатьПредупреждение(, "Договор не создан. Подробности — в сообщениях и журнале регистрации.",, "Ошибка создания!");
		Возврат;
	КонецЕсли;

	Записать(); // сохраняем ссылку на созданный договор в документе

	Попытка
		ПолучитьПрикрепитьДоговоры_PDF_Word();
		ПоказатьОповещениеПользователя("Создано!", ,
			"", БиблиотекаКартинок.Информация32x32, СтатусОповещенияПользователя.Информация);
	Исключение
		ПоказатьПредупреждение(, ОписаниеОшибки(),, "Не создано");
		СообщитьОшибкуВЖурналРегистрации("СоздатьДоговор(Команда)", ОписаниеОшибки());
	КонецПопытки;

	ОбновитьОтображениеДанных();

Функция СоздатьДоговорНаСервере(нКонтрагентСсылка)

	Объект.ИНН = Сервисный.УдалитьИзСтрокиНеЦифры(Объект.ИНН);

	Если НайтиДействующийОткрытыйДоговор(Объект.ИНН) <> Неопределено Тогда
		Возврат Неопределено;
	КонецЕсли;

	// Проверка заполнения обязательных реквизитов
	НезаполненныеРеквизиты = Новый Массив;
	Если Не ЗначениеЗаполнено(нКонтрагентСсылка) Тогда
		НезаполненныеРеквизиты.Добавить("-- Контрагент (владелец договора)");
	КонецЕсли;
	Если Не ЗначениеЗаполнено(Объект.ОтветственныйДоговораКонтрагента) Тогда
		НезаполненныеРеквизиты.Добавить("-- Ответственный за договор");
	КонецЕсли;
	Если Не ЗначениеЗаполнено(Объект.УсловияДоговора) Тогда
		НезаполненныеРеквизиты.Добавить("-- Условия договора");
	КонецЕсли;
	Если Не ЗначениеЗаполнено(Объект.ПроцентИлиСуммаУсловия) Тогда
		НезаполненныеРеквизиты.Добавить("-- Процент или сумма условия");
	КонецЕсли;

	Если НезаполненныеРеквизиты.Количество() > 0 Тогда
		Сообщить("- Не заполнены обязательные реквизиты для создания договора: "
			+ СтрСоединить(НезаполненныеРеквизиты, ", "));
		Возврат Неопределено;
	КонецЕсли;

	НачатьТранзакцию();
	Попытка

		нДоговор = Справочники.ДоговорыКонтрагентов.СоздатьЭлемент();

		нДоговор.Автор = ПараметрыСеанса.ТекущийПользователь;
		нДоговор.ВидДоговора = Перечисления.ВидыДоговоровКонтрагентов.СПоставщиком;
		нДоговор.ТипДоговора = Справочники.ТипыДоговоров.НайтиПоНаименованию(ИмяТипаДоговора());
		нДоговор.Ответственный = Объект.ОтветственныйДоговораКонтрагента;
		нДоговор.ДатаНачала = ТекущаяДата();
		нДоговор.ДатаСоздания = ТекущаяДата();
		нДоговор.Комментарий = Объект.КомментарийКДоговоруКонтрагента;

		нДоговор.ВалютаВзаиморасчетов = ПараметрыСеанса.ВалютаУправленческогоУчета;
		Если ЗначениеЗаполнено(Объект.Город) Тогда
			нДоговор.Город = Объект.Город;
		КонецЕсли;

		нДоговор.Владелец = нКонтрагентСсылка;
		нДоговор.ДокументОснование = Объект.Ссылка;
		нДоговор.КонтрагентСтр = нКонтрагентСсылка.Наименование;

		нДоговор.СуммаВходящейУслуги = 0;
		нДоговор.СуммаДоговора = Объект.ПроцентИлиСуммаУсловия;

		нДоговор.Организация = Справочники.Организации.НайтиПоРеквизиту("ИНН", ИННОрганизацииВладельца());
		нДоговор.Подразделение = Справочники.Подразделения.НайтиПоНаименованию(ИмяПодразделенияПоУмолчанию());
		нДоговор.ЭтоДопсоглашение = Ложь;
		нДоговор.НестандартныйНомер = Ложь;

		// Доп. реквизит «Дилер: ПОСТАВЩИКИ»
		нДопДилер = нДоговор.ДополнительныеРеквизиты.Добавить();
		нДопДилер.Свойство = ПланыВидовХарактеристик.ДополнительныеРеквизитыИСведения.НайтиПоНаименованию("Дилер");
		нДопДилер.Значение = Справочники.ЗначенияСвойствОбъектов.НайтиПоНаименованию("ПОСТАВЩИКИ");

		ЗаполнитьСтороныДоговора(нДоговор, нКонтрагентСсылка);

		нДоговор.Записать(); // первая запись — получаем код для нумерации

		нДоговор.НомерДоговора = ДоговораСервер.УстановитьНомерДоговора(
			нДоговор.Подразделение, нДоговор.ТипДоговора, нДоговор.Код);
		нДоговор.Наименование = нДоговор.ТипДоговора.Наименование
			+ " №" + Строка(нДоговор.НомерДоговора)
			+ " от " + Формат(нДоговор.ДатаНачала, "ДФ=dd.MM.yyyy");

		нДоговор.Записать();

		// Основной договор у контрагента
		Попытка
			нКонтрагентОбъект = нКонтрагентСсылка.ПолучитьОбъект();
			нКонтрагентОбъект.ОсновнойДоговор = нДоговор.Ссылка;
			нКонтрагентОбъект.Записать();
		Исключение
			ЗаписатьОшибкуВЖурнал("СоздатьДоговорНаСервере (основной договор)", ОписаниеОшибки());
		КонецПопытки;

		// Элемент дилера в папке «РАЗОБРАТЬ!»
		Если ЗначениеЗаполнено(Объект.ИНН) И Не ЕстьЭлементДилераПоИНН(Объект.ИНН) Тогда
			СоздатьЭлементДилера();
		КонецЕсли;

		Объект.ДоговорКонтрагента = нДоговор.Ссылка;

		ЗафиксироватьТранзакцию();
		Возврат нДоговор.Ссылка;

	Исключение
		ОтменитьТранзакцию();
		ТекстОшибки = ОписаниеОшибки();
		Сообщить("- Ошибка создания договора контрагента" + Символы.ПС + ТекстОшибки, СтатусСообщения.ОченьВажное);
		ЗаписатьОшибкуВЖурнал("СоздатьДоговорНаСервере", ТекстОшибки);
		Возврат Неопределено;
	КонецПопытки;

Функция ПредставлениеАдресаИзДанных(Данные, СИндексомИРегионом)

	Если СИндексомИРегионом Тогда
		Результат = Данные.Индекс + ", " + Данные.Регион + ", ";
	КонецЕсли;

	Результат = Результат
		+ ?(Данные.Город <> "", "г. ", "") + Данные.Город
		+ Данные.тНаселенПункт + " " + Данные.НаселенПункт + ?(Данные.НаселенПункт <> "", ", ", "")
		+ Данные.тУлица + " " + Данные.Улица
		+ ", " + Данные.тЗдание + " " + Данные.Здание
		+ ", " + Данные.тПомещЗдания + " " + Данные.ПомещЗдания;

	Если ЗначениеЗаполнено(Данные.ПомещКвартиры) Тогда
		Результат = Результат + ", " + Данные.тПомещКвартиры + " " + Данные.ПомещКвартиры;
	КонецЕсли;

	Возврат Результат;

// Создание контрагента по реквизитам из гос. реестра.
&НаСервере
Функция СоздатьКонтрагента(Реквизиты)

	НезаполненныеРеквизиты = Новый Массив;
	Если Не ЗначениеЗаполнено(Реквизиты.ИНН) Тогда
		НезаполненныеРеквизиты.Добавить("ИНН контрагента");
	КонецЕсли;
	Если Не ЗначениеЗаполнено(Реквизиты.НаименованиеПолное) Тогда
		НезаполненныеРеквизиты.Добавить("Наименование полное");
	КонецЕсли;
	Если Не ЗначениеЗаполнено(Реквизиты.ОГРН) Тогда
		НезаполненныеРеквизиты.Добавить("ОГРН");
	КонецЕсли;

	Если НезаполненныеРеквизиты.Количество() > 0 Тогда
		Сообщение = "- Не заполнены обязательные реквизиты контрагента: "
			+ СтрСоединить(НезаполненныеРеквизиты, ", ");
		Сообщить(Сообщение);
		ВызватьИсключение Сообщение;
	КонецЕсли;

	нКонтрагент = Справочники.Контрагенты.СоздатьЭлемент();

	Если ЗначениеЗаполнено(Объект.АгентскийНДС) Тогда
		допАгентскийНДС = нКонтрагент.ДополнительныеРеквизиты.Добавить();
		допАгентскийНДС.Свойство = ПланыВидовХарактеристик.ДополнительныеРеквизитыИСведения.НайтиПоНаименованию(ИмяСвойстваАгентскийНДС());
		допАгентскийНДС.Значение = Справочники.ЗначенияСвойствОбъектов.НайтиПоНаименованию(Объект.АгентскийНДС);
	КонецЕсли;

	нКонтрагент.ПлательщикНДС = Объект.ПлательщикНДС;
	нКонтрагент.ИНН = Реквизиты.ИНН;
	нКонтрагент.КПП = Реквизиты.КПП;
	нКонтрагент.ОГРН = Реквизиты.ОГРН;

	нКонтрагент.ОрганизационноПравоваяФорма = Реквизиты.ОрганизационноПравоваяФорма;
	Если нКонтрагент.ОрганизационноПравоваяФорма = ПредопределенноеЗначение("Справочник.ОрганизационноПравовыеФормы.ИП") Тогда
		нФизЛицо = НайтиФизЛицоПоИНН(Реквизиты.ИНН);
		Если нФизЛицо.Пустая() Тогда
			нФизЛицо = СоздатьФизЛицо(Ложь);
		КонецЕсли;
		нКонтрагент.ФизЛицо = нФизЛицо;
		Объект.ФизЛицо = нФизЛицо;
	КонецЕсли;

	нКонтрагент.ЮрФизЛицо = Объект.ЮрФизЛицо;

	Попытка
		нКонтрагент.СвСтатус_РО = "На " + Формат(ТекущаяДата(), "ДЛФ=D") + " " + Реквизиты.СвСтатус_РО;
	Исключение
		нКонтрагент.СвСтатус_РО = "";
	КонецПопытки;

	нКонтрагент.Страна = Справочники.КлассификаторСтранМира.НайтиПоНаименованию("РОССИЯ");
	нКонтрагент.Город = Справочники.Города.НайтиПоНаименованию(ИмяПодразделенияПоУмолчанию());
	нКонтрагент.Подразделение = Справочники.Подразделения.НайтиПоНаименованию(ИмяПодразделенияПоУмолчанию());
	нКонтрагент.НаименованиеДляДокументов = Реквизиты.НаименованиеДляДокументов;
	нКонтрагент.НаименованиеПолное = Реквизиты.НаименованиеПолное;
	нКонтрагент.НаименованиеКонтрагента = Реквизиты.НаименованиеПолное;
	нКонтрагент.Наименование = Реквизиты.НаименованиеПолное;
	нКонтрагент.Родитель = Справочники.Контрагенты.НайтиПоНаименованию(ИмяГруппыДилеры());

	СоздатьПапкуДействующегоКлиента(Реквизиты.НаименованиеПолное, Реквизиты.ИНН, Ложь);

	нКонтрагент.Записать();

	Объект.КонтрагентНаименованиеДляДокументов = нКонтрагент.НаименованиеДляДокументов;

	Возврат нКонтрагент.Ссылка;

// Создание физлица (и контрагента-самозанятого) по реквизитам документа.
&НаСервере
Функция СоздатьФизЛицо(СоздаватьКонтрагента)

	НезаполненныеРеквизиты = Новый Массив;
	Если Не ЗначениеЗаполнено(Объект.Фамилия) Тогда
		НезаполненныеРеквизиты.Добавить("Фамилия");
	КонецЕсли;
	Если Не ЗначениеЗаполнено(Объект.Имя) Тогда
		НезаполненныеРеквизиты.Добавить("Имя");
	КонецЕсли;
	Если Не ЗначениеЗаполнено(Объект.Отчество) Тогда
		НезаполненныеРеквизиты.Добавить("Отчество");
	КонецЕсли;
	Если Не ЗначениеЗаполнено(Объект.ИНН) Тогда
		НезаполненныеРеквизиты.Добавить("ИНН");
	КонецЕсли;

	Если НезаполненныеРеквизиты.Количество() > 0 Тогда
		Сообщение = "- Не заполнены обязательные поля физ. лица: "
			+ СтрСоединить(НезаполненныеРеквизиты, ", ");
		Сообщить(Сообщение);
		ВызватьИсключение Сообщение;
	КонецЕсли;

	ЭтоСамозанятый = (Объект.ЮрФизЛицо = Перечисления.ЮридическоеФизическоеЛицо.ФизическоеЛицо);

	// 1. Физлицо
	нФизЛицо = Справочники.ФизическиеЛица.СоздатьЭлемент();
	нФизЛицо.Фамилия = Объект.Фамилия;
	нФизЛицо.ДевичьяФамилия = Объект.ДевичьяФамилия;
	нФизЛицо.Имя = Объект.Имя;
	нФизЛицо.Отчество = Объект.Отчество;
	нФизЛицо.ИНН = Объект.ИНН;
	нФизЛицо.ДатаРождения = Объект.ДатаРождения;
	нФизЛицо.Пол = Объект.Пол;
	нФизЛицо.Наименование = СокрЛП(Объект.Фамилия + " " + Объект.Имя + " " + Объект.Отчество);
	нФизЛицо.Автор = ПараметрыСеанса.ТекущийПользователь;
	нФизЛицо.ДатаСоздания = ТекущаяДата();
	нФизЛицо.ФИОДляПодписи = Объект.ФИОДляПодписи;
	нФизЛицо.ФИОРодительныйПадеж = Объект.ФИОРодительныйПадеж;
	нФизЛицо.КодИМНС = Объект.КодИМНС;
	нФизЛицо.СтраховойНомерПФР = Объект.СтраховойНомерПФР;

	// Группа «Дилеры» в справочнике ФизическиеЛица
	ГруппаДилерыФЛ = Справочники.ФизическиеЛица.НайтиПоНаименованию(ИмяГруппыДилеры());
	Если ГруппаДилерыФЛ.Пустая() Тогда
		НоваяГруппа = Справочники.ФизическиеЛица.СоздатьГруппу();
		НоваяГруппа.Наименование = ИмяГруппыДилеры();
		НоваяГруппа.Записать();
		ГруппаДилерыФЛ = НоваяГруппа.Ссылка;
	КонецЕсли;
	нФизЛицо.Родитель = ГруппаДилерыФЛ;

	// 2. Папка в «Действующих клиентах»
	СоздатьПапкуДействующегоКлиента(нФизЛицо.Наименование, Объект.ИНН, ЭтоСамозанятый);

	нФизЛицо.Записать();
	Объект.ФизЛицо = нФизЛицо.Ссылка;

	// 3. Контактная информация физлица
	нФизЛицоОбъект = нФизЛицо.Ссылка.ПолучитьОбъект();

	Если Не ЗначениеЗаполнено(Объект.ОсновнаяЭлектроннаяПочта) Тогда
		СоздатьКонтактнуюИнформацию(нФизЛицо.Ссылка,
			Перечисления.ТипыКонтактнойИнформации.АдресЭлектроннойПочты, Объект.НаименованиеКИЭлектроннаяПочта);
		Объект.ОсновнаяЭлектроннаяПочта = ЗаполнитьEmailФизЛицаИзБазы(Объект.ИНН);
	КонецЕсли;
	нФизЛицоОбъект.ОсновнаяЭлектроннаяПочта = Объект.ОсновнаяЭлектроннаяПочта;

	Если Не ЗначениеЗаполнено(Объект.ОсновнойТелефон) Тогда
		СоздатьКонтактнуюИнформацию(нФизЛицо.Ссылка,
			Перечисления.ТипыКонтактнойИнформации.Телефон, Объект.ТелефонКИФизЛица);
		Объект.ОсновнойТелефон = ЗаполнитьТелефонФизЛицаИзБазы(Объект.ИНН);
	КонецЕсли;
	нФизЛицоОбъект.ОсновнойТелефон = Объект.ОсновнойТелефон;

	нФизЛицоОбъект.Записать();

	// 4. Контрагент для самозанятого
	нКонтрагент = НайтиКонтрагентаПоИНН(Объект.ИНН);
	Если нКонтрагент <> Неопределено Тогда
		Объект.КонтрагентНаименованиеДляДокументов = нКонтрагент.НаименованиеДляДокументов;
	КонецЕсли;

	Если СоздаватьКонтрагента И нКонтрагент = Неопределено Тогда
		нКонтрагент = СоздатьКонтрагентаСамозанятого(нФизЛицо);
	КонецЕсли;

	// 5. Банковские реквизиты
	ТекущийСчет = НайтиБанковскийСчетПоИНН(Объект.ИНН);
	Если ТекущийСчет = Неопределено И ЗначениеЗаполнено(Объект.НомерСчета) И нКонтрагент <> Неопределено Тогда
		ТекущийСчет = СоздатьБанковскийСчет(нКонтрагент);
	КонецЕсли;
	Если ТекущийСчет <> Неопределено Тогда
		ЗаполнитьРеквизитыИзБанковскогоСчета(ТекущийСчет);
		Объект.ОсновнойБанковскийСчет = ТекущийСчет;
		Если нКонтрагент <> Неопределено Тогда
			нКонтрагентОбъект = нКонтрагент.ПолучитьОбъект();
			нКонтрагентОбъект.ОсновнойБанковскийСчет = ТекущийСчет;
			нКонтрагентОбъект.Записать();
		КонецЕсли;
	КонецЕсли;

	// 6. Паспортные данные и адрес
	Если НайтиИЗаполнитьПаспортныеДанные(Объект.ДокументСерия, Объект.ДокументНомер) = Неопределено Тогда
		СоздатьПаспортныеДанные(нФизЛицо.Ссылка);
		НайтиИЗаполнитьПаспортныеДанные(Объект.ДокументСерия, Объект.ДокументНомер);
	КонецЕсли;

	Если ЗаполнитьАдресФизЛицаИлиИП(Объект.ИНН) = Неопределено Тогда
		СоздатьАдресФизЛица(нФизЛицо.Ссылка);
		ЗаполнитьАдресФизЛицаИлиИП(Объект.ИНН);
	КонецЕсли;

	Возврат нФизЛицо.Ссылка;

// Заполняет документ данными существующего физлица и его связанных объектов.
&НаСервере
Функция ЗаполнитьОбъектДаннымиФизЛица()

	нФизЛицо = НайтиФизЛицоПоИНН(Объект.ИНН);
	Если нФизЛицо.Пустая() Тогда
		ВызватьИсключение "Физ. лицо с ИНН " + Объект.ИНН + " не найдено в базе";
	КонецЕсли;

	нКонтрагент = НайтиКонтрагентаПоИНН(Объект.ИНН);

	// Основные данные физлица
	Объект.ФизЛицо = нФизЛицо;
	ЗаполнитьЗначенияСвойств(Объект, нФизЛицо,
		"Фамилия, Имя, Отчество, ДатаРождения, Пол, ДевичьяФамилия, СтраховойНомерПФР, КодИМНС, УникальныйИдентификатор");

	// Паспортные данные
	Если НайтиИЗаполнитьПаспортныеДанные(Объект.ДокументСерия, Объект.ДокументНомер) = Неопределено Тогда
		СоздатьПаспортныеДанные(нФизЛицо);
		НайтиИЗаполнитьПаспортныеДанные(Объект.ДокументСерия, Объект.ДокументНомер);
	КонецЕсли;

	нФизЛицоОбъект = нФизЛицо.ПолучитьОбъект();

	// Адрес: создаём новый, если нет или отличается от введённого
	тАдресСсылка = НайтиАдресФизЛицаИлиИП(Объект.ИНН);
	АдресИзменился = (тАдресСсылка <> Неопределено И ЗначениеЗаполнено(Объект.Адрес)
		И тАдресСсылка.Наименование <> Объект.Адрес);
	Если тАдресСсылка = Неопределено Или АдресИзменился Тогда
		СоздатьАдресФизЛица(нФизЛицо);
	КонецЕсли;
	тАдресСсылка = ЗаполнитьАдресФизЛицаИлиИП(Объект.ИНН);

	// Телефон: создаём новый, если нет или номер изменился
	тТелефонКИ = НайтиКИВладельцаПоИНН(Объект.ИНН, Перечисления.ТипыКонтактнойИнформации.Телефон);
	ТелефонИзменился = (тТелефонКИ <> Неопределено И ЗначениеЗаполнено(Объект.ТелефонКИФизЛица)
		И тТелефонКИ.ТелефонТолькоЦифры <> Сервисный.УдалитьИзСтрокиНеЦифры(Объект.ТелефонКИФизЛица));
	Если (тТелефонКИ = Неопределено Или ТелефонИзменился) И ЗначениеЗаполнено(Объект.ТелефонКИФизЛица) Тогда
		СоздатьКонтактнуюИнформацию(нФизЛицо, Перечисления.ТипыКонтактнойИнформации.Телефон, Объект.ТелефонКИФизЛица);
	КонецЕсли;
	нФизЛицоОбъект.ОсновнойТелефон = ЗаполнитьТелефонФизЛицаИзБазы(Объект.ИНН);

	// Email: аналогично
	тEmailКИ = НайтиКИВладельцаПоИНН(Объект.ИНН, Перечисления.ТипыКонтактнойИнформации.АдресЭлектроннойПочты);
	EmailИзменился = (тEmailКИ <> Неопределено И ЗначениеЗаполнено(Объект.НаименованиеКИЭлектроннаяПочта)
		И тEmailКИ.Наименование <> Объект.НаименованиеКИЭлектроннаяПочта);
	Если (тEmailКИ = Неопределено Или EmailИзменился) И ЗначениеЗаполнено(Объект.НаименованиеКИЭлектроннаяПочта) Тогда
		СоздатьКонтактнуюИнформацию(нФизЛицо, Перечисления.ТипыКонтактнойИнформации.АдресЭлектроннойПочты, Объект.НаименованиеКИЭлектроннаяПочта);
	КонецЕсли;
	нФизЛицоОбъект.ОсновнаяЭлектроннаяПочта = ЗаполнитьEmailФизЛицаИзБазы(Объект.ИНН);

	нФизЛицоОбъект.Записать();

	// Связанный контрагент
	Если нКонтрагент <> Неопределено Тогда

		нКонтрагентОбъект = нКонтрагент.ПолучитьОбъект();

		Если ЗначениеЗаполнено(нКонтрагентОбъект.НаименованиеДляДокументов) Тогда
			Объект.КонтрагентНаименованиеДляДокументов = нКонтрагентОбъект.НаименованиеДляДокументов;
		КонецЕсли;

		СвойствоАгентскийНДС = ПланыВидовХарактеристик.ДополнительныеРеквизитыИСведения.НайтиПоНаименованию(ИмяСвойстваАгентскийНДС());
		Для Каждого допРекв Из нКонтрагентОбъект.ДополнительныеРеквизиты Цикл
			Если допРекв.Свойство = СвойствоАгентскийНДС Тогда
				Объект.АгентскийНДС = Строка(допРекв.Значение);
				Прервать;
			КонецЕсли;
		КонецЦикла;
		Объект.ПлательщикНДС = нКонтрагентОбъект.ПлательщикНДС;

		Если ЗначениеЗаполнено(тАдресСсылка) Тогда
			нКонтрагентОбъект.ОсновнойАдресДоставки = тАдресСсылка;
		КонецЕсли;

		// Банковский счёт: создаём, если нет или номер изменился
		СсылкаСчета = НайтиБанковскийСчетПоИНН(Объект.ИНН);
		СчетИзменился = (СсылкаСчета <> Неопределено И ЗначениеЗаполнено(Объект.НомерСчета)
			И СсылкаСчета.НомерСчета <> Объект.НомерСчета);
		Если (СсылкаСчета = Неопределено Или СчетИзменился) И ЗначениеЗаполнено(Объект.НомерСчета) Тогда
			СсылкаСчета = СоздатьБанковскийСчет(нКонтрагент);
		КонецЕсли;

		Если СсылкаСчета <> Неопределено Тогда
			ЗаполнитьРеквизитыИзБанковскогоСчета(СсылкаСчета);
			Объект.ОсновнойБанковскийСчет = СсылкаСчета;
			нКонтрагентОбъект.ОсновнойБанковскийСчет = СсылкаСчета;
		КонецЕсли;

		нКонтрагентОбъект.Записать();

	КонецЕсли;

	Возврат нФизЛицо;

Функция СоздатьПаспортныеДанные(ФизЛицоСсылка)

	// Все обязательные поля должны быть заполнены
	Если Не ЗначениеЗаполнено(Объект.ДокументСерия)
		Или Не ЗначениеЗаполнено(Объект.ДокументНомер)
		Или Не ЗначениеЗаполнено(Объект.ДокументДатаВыдачи)
		Или Не ЗначениеЗаполнено(Объект.ДокументКемВыдан)
		Или Не ЗначениеЗаполнено(Объект.ДокументКодПодразделения) Тогда
		Возврат Неопределено;
	КонецЕсли;

	МЗ = РегистрыСведений.ПаспортныеДанныеФизЛиц.СоздатьМенеджерЗаписи();

	МЗ.ДокументВид = Справочники.ДокументыУдостоверяющиеЛичность.НайтиПоНаименованию("Паспорт РФ");
	МЗ.ДокументДатаВыдачи = Объект.ДокументДатаВыдачи;
	МЗ.ДокументКемВыдан = Объект.ДокументКемВыдан;
	МЗ.ДокументКодПодразделения = Объект.ДокументКодПодразделения;
	МЗ.ДокументНомер = Сервисный.УдалитьИзСтрокиНеЦифры(Объект.ДокументНомер);
	// Серия хранится в том же виде, в котором ищется (см. НайтиИЗаполнитьПаспортныеДанные)
	МЗ.ДокументСерия = Сервисный.УдалитьИзСтрокиНеЦифры(Объект.ДокументСерия);
	МЗ.Период = ТекущаяДата();
	МЗ.ФизЛицо = ФизЛицоСсылка;

	Если ЗначениеЗаполнено(Объект.СканированныйДокумент) Тогда
		Попытка
			ДвДанные = ПолучитьИзВременногоХранилища(Объект.СканированныйДокумент);
			Если ДвДанные <> Неопределено Тогда
				МЗ.СканированныйДокумент = СохранитьФайлВСправочник(ДвДанные,
					"ПаспортФЛ" + ФизЛицоСсылка.Наименование + " от " + Строка(ТекущаяДата()) + ".pdf",
					ФизЛицоСсылка,
					Справочники.ВидыФайлов.ДоговорСФаксимиле,
					НайтиКонтрагентаПоИНН(Объект.ИНН));
			КонецЕсли;
		Исключение
			ЗаписатьОшибкуВЖурнал("СоздатьПаспортныеДанные (скан)", ОписаниеОшибки());
		КонецПопытки;
	КонецЕсли;

	МЗ.Записать();

	Возврат Неопределено;

Функция НайтиИлиСоздатьКонтактноеЛицо(Фамилия, Имя, Отчество, Пол, Должность, ФИОРодПадеж = "")

	|ИЗ
	|	Справочник.КонтактныеЛицаКонтрагентов КАК КонтактныеЛицаКонтрагентов

	|	И КонтактныеЛицаКонтрагентов.ПометкаУдаления = ЛОЖЬ";

	Результат = Запрос.Выполнить();

	кон = Справочники.КонтактныеЛицаКонтрагентов.СоздатьЭлемент();

	тВладелец = НайтиКонтрагентаПоИНН(Объект.ИНН);
	Если тВладелец <> Неопределено Тогда
		кон.Владелец = тВладелец;
	КонецЕсли;

	кон.Автор = ПараметрыСеанса.ТекущийПользователь;
	кон.ДатаСоздания = ТекущаяДата();
	кон.ДатаРождения = Объект.ДатаРождения;
	кон.ОсновнойТелефон = Объект.ОсновнойТелефон;

	кон.ОсновнойАдрес = Объект.ОсновнойАдресДоставки;

	ЭтоЮЛ = (Объект.ЮрФизЛицо = Перечисления.ЮридическоеФизическоеЛицо.ЮридическоеЛицо);

	Если ЭтоЮЛ И СтрДлина(Объект.ИНН) = 10 Тогда

// Создание элемента справочника КонтактнаяИнформация
// (вместо четырёх копий CREATE_КИ* в исходном модуле).
&НаСервере
Функция СоздатьКонтактнуюИнформацию(Владелец, ТипКИ, Значение)

	Если Не ЗначениеЗаполнено(Значение) Или Не ЗначениеЗаполнено(Владелец) Тогда
		Возврат Неопределено;
	КонецЕсли;

	нКИ = Справочники.КонтактнаяИнформация.СоздатьЭлемент();
	нКИ.Владелец = Владелец;
	нКИ.Тип = ТипКИ;
	нКИ.ДатаСоздания = ТекущаяДата();
	нКИ.Автор = Объект.Автор;

	Если ТипКИ = Перечисления.ТипыКонтактнойИнформации.Телефон Тогда
		Попытка
			нКИ.ВидКонтакнойИнформации = Справочники.ВидыКонтактнойИнформации.НайтиПоНаименованию("Телефон");
		Исключение
			нКИ.ВидКонтакнойИнформации = Справочники.ВидыКонтактнойИнформацииСтарая.НайтиПоНаименованию("Телефон");
		КонецПопытки;
		нКИ.Телефон = Значение;
		нКИ.ТелефонТолькоЦифры = Сервисный.УдалитьИзСтрокиНеЦифры(Строка(Значение));
		нКИ.ДобавочныйНомер = Объект.ДобавочныйНомер;
		нКИ.КодСтраны = Объект.КодСтраны;
		нКИ.КодГорода = Объект.КодГорода;
		нКИ.Наименование = "Телефон: " + Значение;
	Иначе
		Попытка
			нКИ.ВидКонтакнойИнформации = Справочники.ВидыКонтактнойИнформации.EmailПользователя;
		Исключение
			нКИ.ВидКонтакнойИнформации = Справочники.ВидыКонтактнойИнформацииСтарая.Почта;
		КонецПопытки;
		нКИ.Наименование = Значение;
		нКИ.ПредставлениеКИ = Значение;
	КонецЕсли;

	нКИ.Записать();

	Возврат нКИ.Ссылка;

		Объект.ОсновнойТелефон = КИСсылка;
	КонецЕсли;

	Если ЗначениеЗаполнено(Объект.Email) Тогда
		КИСсылка = НайтиКонтактнуюИнформациюПоЗначению(нКонтрагентСсылка,
			Перечисления.ТипыКонтактнойИнформации.АдресЭлектроннойПочты, Строка(Объект.Email));
		Если КИСсылка = Неопределено Тогда
			КИСсылка = СоздатьКонтактнуюИнформацию(нКонтрагентСсылка,
				Перечисления.ТипыКонтактнойИнформации.АдресЭлектроннойПочты, Объект.Email);
		КонецЕсли;
		Объект.ОсновнаяЭлектроннаяПочта = КИСсылка;
	КонецЕсли;

// Обновляет признак плательщика НДС и доп. реквизит «Агентский НДС» контрагента
// (в исходном модуле блок был скопирован трижды).
&НаСервере
Процедура ОбновитьНДСКонтрагента(нКонтрагентСсылка)

	Если Не ЗначениеЗаполнено(нКонтрагентСсылка) Или Не ЗначениеЗаполнено(Объект.ПлательщикНДС) Тогда
		Возврат;
	КонецЕсли;

	Попытка

		нКонтрагентОбъект = нКонтрагентСсылка.ПолучитьОбъект();
		нКонтрагентОбъект.ПлательщикНДС = Объект.ПлательщикНДС;

		Если ЗначениеЗаполнено(Объект.АгентскийНДС) Тогда

			СвойствоАгентскийНДС = ПланыВидовХарактеристик.ДополнительныеРеквизитыИСведения.НайтиПоНаименованию(ИмяСвойстваАгентскийНДС());
			ЗначениеАгентскогоНДС = Справочники.ЗначенияСвойствОбъектов.НайтиПоНаименованию(Объект.АгентскийНДС);

			СтрокаНДС = Неопределено;
			Для Каждого доп Из нКонтрагентОбъект.ДополнительныеРеквизиты Цикл
				Если доп.Свойство = СвойствоАгентскийНДС Тогда
					СтрокаНДС = доп;
					Прервать;
				КонецЕсли;
			КонецЦикла;

			Если СтрокаНДС = Неопределено Тогда
				СтрокаНДС = нКонтрагентОбъект.ДополнительныеРеквизиты.Добавить();
				СтрокаНДС.Свойство = СвойствоАгентскийНДС;
			КонецЕсли;
			СтрокаНДС.Значение = ЗначениеАгентскогоНДС;

		КонецЕсли;

		нКонтрагентОбъект.Записать();

	Исключение
		Сообщить("Ошибка при обновлении ставки НДС контрагента: " + ОписаниеОшибки());
		ЗаписатьОшибкуВЖурнал("ОбновитьНДСКонтрагента", ОписаниеОшибки());
	КонецПопытки;

		Возврат "" + Формат(Окр(РазмерФайла / (1024 * 1024 * 1024), 2), "ЧРГ=") + " ГБ";

		Возврат "" + Формат(Окр(РазмерФайла / (1024 * 1024 * 1024 * 1024), 2), "ЧРГ=") + " ТБ";

Функция ПривестиКФормату(ЗначениеЗамены, Формат)

		Возврат ЗначениеЗамены;

// Контрагент передаётся явно: серверные переменные модуля формы

Функция СохранитьФайлВСправочник(ДвоичныеДанные, ИмяФайла, ОбъектПривязки, ВидФайла, КонтрагентСсылка)

	ФайлОбъект = Справочники.Файлы.СоздатьЭлемент();
	ФайлОбъект.Наименование = Новый УникальныйИдентификатор();
	ФайлОбъект.ИмяФайла = ИмяФайла;
	ФайлОбъект.РасширениеФайла = ПолучитьРасширение(ИмяФайла);
	ФайлОбъект.Автор = ПараметрыСеанса.Пользователь;
	ФайлОбъект.ДатаСоздания = ТекущаяДата();
	ФайлОбъект.ВидФайла = ВидФайла;
	ФайлОбъект.Объект = ОбъектПривязки;

	Если ЗначениеЗаполнено(КонтрагентСсылка) Тогда
		ФайлОбъект.Контрагент = КонтрагентСсылка;
	ИначеЕсли ЗначениеЗаполнено(Объект.ДоговорКонтрагента) Тогда
		ФайлОбъект.Контрагент = Объект.ДоговорКонтрагента.Владелец;
	КонецЕсли;

	Размер = ДвоичныеДанные.Размер();
	ФайлОбъект.РазмерФайла = Размер;
	ФайлОбъект.ПредставлениеРазмераФайла = ПредставлениеРазмераФайлов(Размер);
	ФайлОбъект.Файл = Неопределено;
	ФайлОбъект.ФайлВыбран = Истина;
	ФайлОбъект.Записать();

	ПутьКФайлуНаДиске = Строка(Константы.ПутьКФайлам.Получить()) + ФайлОбъект.Наименование;
	ДвоичныеДанные.Записать(ПутьКФайлуНаДиске);

	Возврат ФайлОбъект.Ссылка;
